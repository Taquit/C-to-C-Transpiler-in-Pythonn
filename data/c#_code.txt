using System;

namespace TranspiladorPrueba {
    class Program {
        
        static string ObtenerSaludo(string nombre) {
            return "Saludos, " + nombre;
        }

        static int Multiplicar(int baseNum, int multiplicador) {
            return baseNum * multiplicador;
        }

        static void ImprimirConteo(int limite) {
            for (int i = 0; i < limite; i++) {
                Console.WriteLine("Contando");
            }
        }

        static void Main(string[] args) {
            Console.WriteLine("Iniciando prueba completa...");
            
            string miNombre = "Usuario";
            string saludo = ObtenerSaludo(miNombre);
            Console.WriteLine(saludo);
            
            int factor = 5;
            int total = Multiplicar(factor, 2);
            
            if (total == 10) {
                Console.WriteLine("La multiplicacion es correcta");
            } else {
                Console.WriteLine("Error en la logica");
            }

            ImprimirConteo(3);

            float temperatura = 36.5f;
            while (temperatura < 38.0f) {
                temperatura += 1.0f;
                Console.WriteLine("Subiendo temperatura");
            }

            int opcion = 2;
            switch (opcion) {
                case 1:
                    Console.WriteLine("Opcion 1 seleccionada");
                    break;
                case 2:
                    Console.WriteLine("Opcion 2 seleccionada");
                    break;
                default:
                    Console.WriteLine("Opcion desconocida");
                    break;
            }

            Console.WriteLine("Prueba finalizada exitosamente.");
        }
    }
}
